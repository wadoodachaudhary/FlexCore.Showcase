@code {
    private List<VendorRecord> _vendors = [];
    private List<VendorContactRecord> _vendorContacts = [];
    private VendorRecord? _currentVendor;
    private int _vendorIndex = 0;
    private int _activeTab = 0;

    protected override void OnInitialized()
    {
        _vendors = BenchData.Vendors.ToList();
        if (_vendors.Count > 0)
        {
            SetCurrentVendor(0);
        }
    }

    private void SetCurrentVendor(int index)
    {
        _vendorIndex = Math.Clamp(index, 0, _vendors.Count - 1);
        _currentVendor = _vendors[_vendorIndex];
        _vendorContacts = BenchData.VendorContacts
            .Where(c => c.VendorCode == _currentVendor.Vendor_ID)
            .ToList();

        if (_vendorContacts.Count == 0)
        {
            _vendorContacts =
            [
                new VendorContactRecord
                {
                    VendorCode = _currentVendor.Vendor_ID,
                    DisplayName = "Primary Contact",
                    Role = "Branch Manager",
                    WorkPhone = _currentVendor.Phone,
                    Email = $"contact@{_currentVendor?.Vendor_ID?.ToLower() ?? "vendor"}.example",
                    Active = true
                }
            ];
        }
    }

    private void PreviousVendor() => SetCurrentVendor(_vendorIndex - 1);
    private void NextVendor() => SetCurrentVendor(_vendorIndex + 1);

    private const string _razorSourceCode = @"@using Fx.ControlKit
@using Fx.ControlKit.Grid

<!-- PageControl manages multi-tab line-of-business workflows -->
<PageControl NavigationGraph=""@NavigationGraph"">
    <div class=""workspace-tabs"">
        <!-- Tab 1: Profile Form -->
        <TextBoxControl Value=""@Vendor.Vendor_ID"" Enabled=""false"" />
        <TextBoxControl Value=""@Vendor.Vendor_Name"" ValueChanged=""OnNameChanged"" />

        <!-- Tab 2: Embedded Sub-Grid -->
        <GridControl TValue=""VendorContactRecord"" DataSource=""@Contacts"">
            <GridColumnsBase>
                <GridColumn Field=""@nameof(VendorContactRecord.DisplayName)"" HeaderText=""Contact"" />
                <GridColumn Field=""@nameof(VendorContactRecord.WorkPhone)"" HeaderText=""Phone"" />
            </GridColumnsBase>
        </GridControl>
    </div>
</PageControl>";

    private const string _csharpSourceCode = @"namespace FlexCore.Showcase.Data;

public sealed class VendorRecord
{
    public string? Vendor_ID { get; set; }
    public string? Vendor_Name { get; set; }
    public string? Addr1 { get; set; }
    public string? City { get; set; }
    public string? State { get; set; }
    public string? Zip { get; set; }
    public string? Phone { get; set; }
    public bool? GLInsRequired { get; set; }
    public bool? WCInsRequired { get; set; }
}";
}
